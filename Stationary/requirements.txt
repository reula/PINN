#   the image ships a CUDA toolkit / module, and skip the ~3 GB of pip wheels):
#     pip install "jax[cuda12-local]==0.11.1" && pip install -r requirements.txt
#

#     python -c "import jax; print(jax.__version__, jax.devices())"
